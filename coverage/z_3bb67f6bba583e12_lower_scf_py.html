3"""Lower the builtin ``scf`` dialect into ``q1_scf``. 

4 

5This pass converts standard MLIR structured control flow (:mod:`xdsl.dialects.scf`) 

6into the Q1-sequencer-specific structured control flow dialect ``q1_scf``. 

7 

8Motivation 

9---------- 

10The frontend importer produces ``scf.for`` loops to represent shot iteration and 

11similar counted patterns. Before the ``q1_scf`` pipeline can lower those loops to 

12hardware, the generic ``scf`` containers must first be mapped onto their Q1 

13counterparts. 

14 

15For loops 

16--------- 

17``scf.for %i = %lb to %ub step %step iter_args(%a, ...) { ... scf.yield ... }`` 

18maps onto ``q1_scf.for %n : q1.reg iter_args(%a, ...) { ... q1_scf.yield ... }`` 

19where ``%n`` is loaded with the iteration count ``ceil((ub - lb) / step)`` by a 

20``q1.ir.move`` instruction inserted immediately before the loop. 

21 

22The ``scf`` induction variable counts *up* from ``lb`` toward ``ub``; the ``q1_scf`` 

23induction counter counts *down* from ``n`` toward zero. When the loop body uses the 

24induction variable, arithmetic instructions are inserted at the top of the body to 

25reconstruct the ascending value ``lb + (count - counter) * step``. 

26 

27Bounds that are not ``arith.constant`` integer literals, non-positive step values, 

28zero-trip counts, and loop-carried values whose type is not ``q1.reg`` are also 

29rejected. 

30 

31Stubs 

32----- 

33``scf.if`` and ``scf.while`` are not yet supported. The corresponding rewrite 

34patterns raise :class:`~xdsl.utils.exceptions.PassFailedException` immediately. 

35 

36Reference: https://docs.qblox.com/en/main/products/qblox_instruments/q1/index.html 

37"""

73def _get_static_integer(value: SSAValue) -> int | None: 

74 """Return the integer value of a static ``arith.constant`` result, or ``None``. 

75 

76 :returns: The integer constant, or ``None`` if not a static integer constant. 

77 """ 

78 if not isinstance(value, OpResult): 

79 return None 

80 if not isinstance(value.op, ArithConstantOp): 

81 return None 

82 attr = value.op.value 

83 if not isinstance(attr, IntegerAttr): 

84 return None 

85 return int(attr.value.data)

88def _remap_induction_uses( 

89 induction_var: SSAValue, 

90 countdown: SSAValue, 

91 lb: int, 

92 count: int, 

93 step: int, 

94 body_block, 

95 rewriter: PatternRewriter, 

96) -> None: 

97 """Replace uses of the ``scf.for`` induction variable with the ascending index. 

98 

99 Inserts arithmetic at the top of *body_block* to compute 

100 ``lb + (count - countdown) * step`` from the ``q1_scf.for`` countdown arg. 

101 """ 

102 # ~countdown + (count + 1) == count - countdown (two's complement) 

103 not_op = NotRsRdOp(countdown, IntRegisterType.unallocated()) 

104 rewriter.insert_op(not_op, InsertPoint.at_start(body_block)) 

105 

106 if step == 1: 

107 ascending = AddRsImmRdOp( 

108 not_op.rd, SU32Imm(lb + count + 1), IntRegisterType.unallocated() 

109 ) 

110 rewriter.insert_op(ascending, InsertPoint.after(not_op)) 

111 induction_var.replace_all_uses_with(ascending.rd) 

112 else: 

113 count_minus_c = AddRsImmRdOp( 

114 not_op.rd, SU32Imm(count + 1), IntRegisterType.unallocated() 

115 ) 

116 rewriter.insert_op(count_minus_c, InsertPoint.after(not_op)) 

117 scaled = Muls16RsImmRdOp( 

118 count_minus_c.rd, SI16Imm(step), IntRegisterType.unallocated() 

119 ) 

120 rewriter.insert_op(scaled, InsertPoint.after(count_minus_c)) 

121 if lb == 0: 

122 induction_var.replace_all_uses_with(scaled.rd) 

123 else: 

124 shifted = AddRsImmRdOp(scaled.rd, SU32Imm(lb), IntRegisterType.unallocated()) 

125 rewriter.insert_op(shifted, InsertPoint.after(scaled)) 

126 induction_var.replace_all_uses_with(shifted.rd)

129class ForLowering(RewritePattern): 

130 """Lower ``scf.for`` to ``q1_scf.for``. 

131 The iteration count ``ceil((ub - lb) / step)`` is materialised as a ``q1.ir.move`` 

132 inserted immediately before the loop. The body induction argument is retyped 

133 from ``index`` to ``!q1.reg``, and ``scf.yield`` is replaced by ``q1_scf.yield``. 

134 When the body uses the induction variable, arithmetic is inserted at the top of 

135 the body block to reconstruct the ascending index from the countdown counter. 

136 

137 Example (``lb=0``, ``ub=10``, ``step=2``, ``count=5``; body uses ``%i``): 

138 

139 .. code-block:: mlir 

140 

141 // before 

142 %lb = arith.constant 0 : index 

143 %ub = arith.constant 10 : index 

144 %step = arith.constant 2 : index 

145 scf.for %i = %lb to %ub step %step { 

146 ... uses of %i ... 

147 scf.yield 

148 } 

149 

150 // after 

151 %lb = arith.constant 0 : index 

152 %ub = arith.constant 10 : index 

153 %step = arith.constant 2 : index 

154 %n = q1.ir.move () <{imm = #q1.su32_imm<5>}> : () -> !q1.reg 

155 q1_scf.for %n : !q1.reg { 

156 ^bb0(%counter: !q1.reg): 

157 %not_c = q1.rr.not (%counter) : (!q1.reg) -> !q1.reg 

158 %diff = q1.rir.add (%not_c) <{imm = #q1.su32_imm<6>}> : (!q1.reg) -> !q1.reg 

159 %i = q1.rir.muls16 (%diff) <{imm = #q1.si16_imm<2>}> : (!q1.reg) -> !q1.reg 

160 ... uses of %i (now !q1.reg, ascending 0, 2, 4, 6, 8) ... 

161 q1_scf.yield 

162 } 

163 """ 

164 

165 @op_type_rewrite_pattern 

166 def match_and_rewrite(self, op: ScfForOp, rewriter: PatternRewriter) -> None: 

167 lb_int = _get_static_integer(op.lb) 

168 ub_int = _get_static_integer(op.ub) 

169 step_int = _get_static_integer(op.step) 

170 

171 if lb_int is None: 

172 raise PassFailedException( 

173 f"{op.name}: lower bound is not a static integer constant" 

174 ) 

175 if ub_int is None: 

176 raise PassFailedException( 

177 f"{op.name}: upper bound is not a static integer constant" 

178 ) 

179 if step_int is None: 

180 raise PassFailedException(f"{op.name}: step is not a static integer constant") 

181 if step_int <= 0: 

182 raise PassFailedException(f"{op.name}: step must be positive, got {step_int}") 

183 

184 # ceiling division: -((-diff) // step) avoids float and matches scf.for semantics 

185 count = -(-(ub_int - lb_int) // step_int) 

186 if count <= 0: 

187 raise PassFailedException( 

188 f"{op.name}: iteration count {count} is not positive;" 

189 " zero-trip loops cannot be represented by q1_scf.for" 

190 ) 

191 

192 for i, iter_arg in enumerate(op.iter_args): 

193 if not isinstance(iter_arg.type, IntRegisterType): 

194 raise PassFailedException( 

195 f"{op.name}: iter_arg {i} has type {iter_arg.type!r};" 

196 " only q1.reg loop-carried values are supported" 

197 ) 

198 

199 body_block = list(op.body.blocks)[0] 

200 induction_var = body_block.args[0] 

201 

202 new_induction = body_block.insert_arg(IntRegisterType.unallocated(), 0) 

203 if any(True for _ in induction_var.uses): 

204 _remap_induction_uses( 

205 induction_var, new_induction, lb_int, count, step_int, body_block, rewriter 

206 ) 

207 body_block.erase_arg(induction_var) 

208 

209 body_yield = body_block.last_op 

210 if not isinstance(body_yield, ScfYieldOp): 

211 raise PassFailedException( 

212 f"{op.name}: body region terminator must be scf.yield, got" 

213 f" {body_yield.name if body_yield is not None else 'nothing'}" 

214 ) 

215 yield_values = list(body_yield.operands) 

216 rewriter.erase_op(body_yield) 

217 rewriter.insert_op(YieldOp(*yield_values), InsertPoint.at_end(body_block)) 

218 

219 move_op = MoveImmRdOp(SU32Imm(count), IntRegisterType.unallocated()) 

220 rewriter.insert_op(move_op, InsertPoint.before(op)) 

221 

222 # Detach body_block from op.body before passing it to ForOp; xDSL's 

223 # Region.__init__ raises if a block is already owned by another region. 

224 op.body.detach_block(body_block) 

225 new_for = ForOp(move_op.rd, list(op.iter_args), body_block) 

226 rewriter.replace_op(op, new_for)

255class LowerScfToQ1ScfPass(ModulePass): 

256 """Lower every ``scf`` operation in the module to its ``q1_scf`` equivalent. 

257 

258 Currently only ``scf.for`` is fully implemented. ``scf.if`` and ``scf.while`` 

259 raise :class:`~xdsl.utils.exceptions.PassFailedException` if encountered. 

260 """ 

261 

262 name = "lower-scf-to-q1-scf" 

263 

264 def apply(self, ctx: Context, op: ModuleOp) -> None: 

265 PatternRewriteWalker( 

266 GreedyRewritePatternApplier( 

267 [ForLowering(), WhileLowering(), IfLowering()], 

268 dce_enabled=False, 

269 ) 

270 ).rewrite_module(op)
