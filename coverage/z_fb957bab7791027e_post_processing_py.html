10UNMAPPED_STATE_LABEL = -1 

11"""The state label a maximum-likelihood discriminator emits when the winning normalised 

12likelihood falls below ``p_min``, i.e. when no centroid can be assigned confidently."""

15@dataclass(frozen=True, slots=True) 

16class DiscriminateData: 

17 """The max-likelihood payload a channel is calibrated with. 

18 

19 :ivar noise_est: Global Gaussian noise variance used in likelihood evaluation. 

20 :ivar p_min: Minimum normalised likelihood required for acceptance. When non-zero, a 

21 shot whose winning likelihood falls below it is emitted as 

22 :data:`UNMAPPED_STATE_LABEL`. 

23 :ivar state_centroids: The calibrated IQ-plane centroids, in calibration order. A 

24 discriminator labels a shot by the **position** of the centroid it is nearest. 

25 """ 

26 

27 noise_est: float 

28 p_min: float 

29 state_centroids: tuple[complex, ...]

32def _discriminate_data(method: MaxLikelihoodMethodData) -> DiscriminateData: 

33 """Extract the discriminate payload for a calibrated max-likelihood method. 

34 

35 Every calibrated centroid is kept, in calibration order. Discrimination assigns an 

36 IQ value to whichever centroid it is nearest. 

37 

38 :param method: The channel's calibrated max-likelihood method. 

39 :returns: The channel's :class:`DiscriminateData`. 

40 """ 

41 return DiscriminateData( 

42 noise_est=method.noise_est, 

43 p_min=method.p_min, 

44 state_centroids=tuple(params.location for _, params in method.states), 

45 )

48def _disallowed_labels(method: MaxLikelihoodMethodData) -> set[int]: 

49 """Map a calibrated method onto the state labels post-selection must discard. 

50 

51 A discriminator built from this method labels its centroids **positionally**: the 

52 ``i``-th centroid in calibration order is emitted as label ``i``. 

53 

54 When ``p_min`` is non-zero the discriminator can additionally decline to classify a 

55 shot, emitting :data:`UNMAPPED_STATE_LABEL`. Those shots are ambiguous by 

56 construction and are always discarded. 

57 

58 :param method: The channel's calibrated max-likelihood method. 

59 :returns: The set of emitted state labels that should be post-selected out. 

60 """ 

61 labels = {index for index, (key, _) in enumerate(method.states) if key < 0} 

62 if method.p_min > 0.0: 

63 labels.add(UNMAPPED_STATE_LABEL) 

64 return labels

67@dataclass(frozen=True, slots=True, kw_only=True) 

68class PostProcessingView: 

69 """Post-processing configuration derived from a :class:`CanonicalSystemData`. 

70 

71 A discriminator only assigns an IQ value to its nearest calibrated centroid, 

72 labelling centroids by their position in calibration order; it carries no notion 

73 of which outcomes are wanted. 

74 The calibrated integer keys, which encode exactly that, are resolved here into the 

75 emitted labels post-selection must discard. 

76 

77 :ivar channel_to_disallowed_states: Mapping from logical channel identifier 

78 (``mode.channel_id`` in the canonical schema) to the set of **emitted state 

79 labels** that should be post-processed out. These are positions in the 

80 ``channel_to_discriminate_data`` centroid ordering, not the calibrated keys they 

81 were derived from, plus :data:`UNMAPPED_STATE_LABEL` when ``p_min`` is non-zero. 

82 Channels with nothing to discard are omitted. 

83 :ivar channel_to_discriminate_data: Mapping from logical channel identifier 

84 (``mode.channel_id`` in the canonical schema) to the single 

85 :class:`DiscriminateData` calibrated for that channel. Present for every channel 

86 carrying a max-likelihood calibration, independently of whether that channel has 

87 anything to post-select out. 

88 :ivar known_channel_ids: All channel IDs present in the canonical system data. 

89 """ 

90 

91 channel_to_disallowed_states: dict[str, set[int]] 

92 known_channel_ids: frozenset[str] 

93 channel_to_discriminate_data: dict[str, DiscriminateData] 

94 

95 @classmethod 

96 def derive(cls, parent: CanonicalSystemData) -> "PostProcessingView": 

97 """Builds the post-processing configuration from a canonical hardware model. 

98 

99 Walks all qubits and their modes, keyed directly by ``mode.channel_id``. Several 

100 modes may share a channel, and a channel resolves to exactly one max-likelihood 

101 method: modes sharing a channel must carry structurally identical calibration, 

102 otherwise there is no basis for choosing between them and this raises. Disallowed 

103 labels are aligned with the centroid ordering they index into. 

104 

105 Channels with no max-likelihood calibration are absent from both mappings, as 

106 there is nothing to discriminate against. Every channel that *does* carry a 

107 calibration gets a discriminate payload, whether or not it has anything to 

108 post-select out; only :attr:`channel_to_disallowed_states` additionally omits 

109 channels whose disallowed-label set is empty. 

110 

111 :param parent: The canonical hardware model to derive post-processing from. 

112 :returns: The derived post-processing configuration for the system. 

113 :raises ValueError: If modes sharing a channel carry different max-likelihood 

114 methods. 

115 """ 

116 channel_to_method: dict[str, MaxLikelihoodMethodData] = {} 

117 for qubit in parent.qubits: 

118 for mode in qubit.modes: 

119 method = mode.post_process_method 

120 if not isinstance(method, MaxLikelihoodMethodData): 

121 continue 

122 existing = channel_to_method.setdefault(mode.channel_id, method) 

123 if existing != method: 

124 raise ValueError( 

125 f"Modes on channel '{mode.channel_id}' carry different " 

126 "max-likelihood methods." 

127 ) 

128 

129 return cls( 

130 channel_to_disallowed_states={ 

131 channel_id: labels 

132 for channel_id, method in channel_to_method.items() 

133 if (labels := _disallowed_labels(method)) 

134 }, 

135 channel_to_discriminate_data={ 

136 channel_id: _discriminate_data(method) 

137 for channel_id, method in channel_to_method.items() 

138 }, 

139 known_channel_ids=frozenset(ch.id for ch in parent.channels), 

140 )

142 def disallowed_states_for_channel(self, channel_id: str) -> set[int]: 

143 """Returns the disallowed emitted state labels for a given logical channel. 

144 

145 :param channel_id: The logical channel identifier to look up. 

146 :returns: The set of disallowed state labels, or an empty set if the channel has 

147 nothing to post-select out. 

148 """ 

149 return self.channel_to_disallowed_states.get(channel_id, set())

151 def unmatched_channels(self, channel_ids: list[str]) -> list[str]: 

152 """Return the channel IDs that are not known to the system data at all. 

153 

154 A channel is considered *unmatched* only when it is entirely absent from the 

155 canonical system data (i.e. not in :attr:`known_channel_ids`). 

156 

157 :param channel_ids: The logical channel identifiers to check. 

158 :returns: A list of channel IDs from *channel_ids* that are absent from 

159 :attr:`known_channel_ids`. Empty if all are present. 

160 """ 

161 return [ch for ch in channel_ids if ch not in self.known_channel_ids]
