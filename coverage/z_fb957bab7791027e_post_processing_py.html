13def _disallowed_states_for_mode(mode: ModeData) -> set[int]: 

14 """Extracts the disallowed integer state labels for a mode. 

15 

16 Only :class:`MaxLikelihoodMethodData` carries explicit integer state labels. 

17 Negative keys in ``states`` are the disallowed/background states per the schema 

18 convention. :class:`LinearMapToRealMethodData` produces a thresholded real value 

19 rather than a discrete state key, so it contributes no disallowed states. 

20 

21 :param mode: The mode to extract disallowed states from. 

22 :returns: The set of integer state labels that should be post-selected out. 

23 """ 

24 match mode.post_process_method: 

25 case MaxLikelihoodMethodData(states=states): 

26 return {key for key, _ in states if key < 0} 

27 case _: 

28 return set()

31@dataclass(frozen=True, slots=True, kw_only=True) 

32class PostProcessing: 

33 """Post-selection configuration derived from a :class:`CanonicalSystemData`. 

34 

35 Provides a mapping from logical channel identifier to the set of integer 

36 state labels that should be discarded during post-selection. Only channels whose mode 

37 has at least one disallowed state are included. 

38 

39 :ivar channel_to_disallowed_states: Mapping from logical channel identifier 

40 (``mode.channel_id`` in the canonical schema) to the set of integer state 

41 labels that should be post-selected out. 

42 :ivar known_channel_ids: All channel IDs present in the canonical system data. 

43 """ 

44 

45 channel_to_disallowed_states: dict[str, set[int]] 

46 known_channel_ids: frozenset[str] 

47 

48 @classmethod 

49 def derive(cls, parent: CanonicalSystemData) -> "PostProcessing": 

50 """Builds the post-selection configuration from a canonical hardware model. 

51 

52 Walks all qubits and their modes and collects disallowed integer state labels 

53 keyed directly by ``mode.channel_id``. Channels with no disallowed states are 

54 omitted. 

55 

56 :param parent: The canonical hardware model to derive post-selection from. 

57 :returns: The derived post-selection configuration for the system. 

58 """ 

59 channel_to_disallowed_states: dict[str, set[int]] = defaultdict(set) 

60 for qubit in parent.qubits: 

61 for mode in qubit.modes: 

62 if disallowed := _disallowed_states_for_mode(mode): 

63 channel_to_disallowed_states[mode.channel_id].update(disallowed) 

64 return cls( 

65 channel_to_disallowed_states=dict(channel_to_disallowed_states), 

66 known_channel_ids=frozenset(ch.id for ch in parent.channels), 

67 )

69 def disallowed_states_for_channel(self, channel_id: str) -> set[int]: 

70 """Returns the disallowed integer state labels for a given logical channel. 

71 

72 :param channel_id: The logical channel identifier to look up. 

73 :returns: The set of disallowed integer state labels, or an empty set if the channel 

74 has no disallowed states. 

75 """ 

76 return self.channel_to_disallowed_states.get(channel_id, set())

78 def unmatched_channels(self, channel_ids: list[str]) -> list[str]: 

79 """Return the channel IDs that are not known to the system data at all. 

80 

81 A channel is considered *unmatched* only when it is entirely absent from the 

82 canonical system data (i.e. not in :attr:`known_channel_ids`). 

83 

84 :param channel_ids: The logical channel identifiers to check. 

85 :returns: A list of channel IDs from *channel_ids* that are absent from 

86 :attr:`known_channel_ids`. Empty if all are present. 

87 """ 

88 return [ch for ch in channel_ids if ch not in self.known_channel_ids]
