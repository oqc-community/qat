13@irdl_attr_definition 

14class RecordType(ParametrizedAttribute, TypeAttribute): 

15 """A type that carries a results record that can be added to a results collection. 

16 

17 This is used to represent a single record of results, e.g., from a single shot. It has 

18 the semantics of a dictionary, where the entries are accessed by string keys, and it can 

19 store arbitrary data types. 

20 

21 :class:`RecordType` is modelled to be immutable, and must be complete when constructed. 

22 

23 :ivar schema: The schema of the record, which defines the keys and types of the entries 

24 in the record. 

25 """ 

26 

27 name = "results.record" 

28 schema: RecordSchemaAttr 

29 

30 def __init__(self, schema: RecordSchemaAttr): 

31 """Initializes a record type with a schema.""" 

32 super().__init__(schema)

35@irdl_attr_definition 

36class ResultsArrayType(ParametrizedAttribute, TypeAttribute): 

37 """A type that represents an array of results, which can be added to and filtered with 

38 given operations. 

39 

40 This is used to represent a collection of results that can be indexed into, e.g., a 

41 register of classical bits measured from a quantum circuit. It takes standard array 

42 semantics, holding an ordered list of results. 

43 

44 :class:`ResultsArrayType` is modelled to be immutable, and must be complete when 

45 constructed. 

46 

47 :ivar type: The type of the entries in the array. 

48 :ivar size: The size of the array. The special value ``DYNAMIC_INDEX`` indicates 

49 runtime-dynamic size. 

50 """ 

51 

52 name = "results.array" 

53 type: TypeAttribute = param_def(_TypeConstraint()) 

54 size: IntAttr 

55 

56 def __init__( 

57 self, 

58 type_: TypeAttribute, 

59 size: IntAttr, 

60 ): 

61 """Initializes an array type with element type and size.""" 

62 super().__init__(type_, size) 

63 

64 @classmethod 

65 def dynamic_size(cls, type_: TypeAttribute) -> "ResultsArrayType": 

66 """Construct a :class:`ResultsArrayType` with a dynamic size."""

70@irdl_attr_definition 

71class ResultsCollectionType(ParametrizedAttribute, TypeAttribute): 

72 """A type that represents a collection of results, which can be added to and filtered 

73 with given operations. 

74 

75 The collection is modelled as a collection of data which are referenced by two 

76 identifiers: 

77 

78 * An integer index, which for example, might refer to the shot number. 

79 * A string key, which for example, might refer to the name of the result. 

80 

81 The actual data type is not specified to be index-major or key-major. This allows us to 

82 support flexible results acquisition. For example, you could treat this as a dictionary 

83 of arrays, and append a result to each array for each shot. Or you could treat this as 

84 a list of records, and append a new record for each shot. 

85 

86 :ivar schema: The schema of the records in the collection, which defines the keys and 

87 types. 

88 :ivar size: The size of the collection. The special value ``DYNAMIC_INDEX`` indicates 

89 runtime-dynamic size. 

90 """ 

91 

92 name = "results.collection" 

93 schema: RecordSchemaAttr 

94 size: IntAttr 

95 

96 def __init__( 

97 self, 

98 schema: RecordSchemaAttr, 

99 size: IntAttr, 

100 ): 

101 """Initializes a collection type with schema and size.""" 

102 super().__init__(schema, size) 

103 

104 @classmethod 

105 def dynamic_size(cls, schema: RecordSchemaAttr) -> "ResultsCollectionType": 

106 """Construct a :class:`ResultsCollectionType` with a dynamic size."""
