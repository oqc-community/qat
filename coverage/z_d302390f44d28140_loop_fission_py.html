3"""Fission a shot loop into one copy per frame lineage. 

4 

5A shot loop drives every frame in the program, so it belongs to every lineage at once and 

6cannot be attributed to a single outlined sequence. Qblox runs one program per sequencer, 

7each with its own loop, so the loop is replicated. 

8 

9Only ``scf.for`` is handled. It is the sole region-bearing operation the importer puts 

10in a Pulse entry block, so it is the only shape that needs splitting today. Anything else 

11enclosing more than one lineage is rejected rather than guessed at, since splitting it 

12would need rules for its regions and block arguments that nothing yet exercises. 

13 

14The copy carries no values between iterations. An ``scf.for`` can pass a value from one 

15iteration to the next through ``iter_args``, and the importer uses that to accumulate 

16acquisitions into a results array. A sequencer has no equivalent: acquisitions land in 

17hardware bins, and the bin index is recomputed from the loop counter when the acquire is 

18lowered. The array is bookkeeping with no hardware counterpart, so the copy simply does 

19not have it. 

20""" 

21

40def is_results_op(op: Operation) -> bool: 

41 """Return whether ``op`` is a results-dialect operation. 

42 

43 Results operations are bookkeeping with no hardware counterpart, so they are dropped 

44 from emitted sequences. Membership is tested against the dialect's registered 

45 operations. 

46 

47 :param op: Operation to classify. 

48 :returns: ``True`` if ``op`` belongs to the results dialect. 

49 """ 

50 return type(op) in Results.operations

53def fission_for_lineage( 

54 op: Operation, 

55 lineage: FrameLineage, 

56 analysis: FrameLineageAnalysis, 

57 value_mapper: dict[SSAValue, SSAValue], 

58) -> ForOp: 

59 """Build a copy of an ``scf.for`` with only the contents belonging to ``lineage``. 

60 

61 :param op: Shot loop enclosing the work of one or more lineages. 

62 :param lineage: Lineage whose work the returned copy carries. 

63 :param analysis: Analysis providing per-operation lineage ownership. 

64 :param value_mapper: Clone mapping for the partition, used to resolve operands 

65 defined by entry-block operations already copied into this sequence. 

66 :returns: A newly built ``scf.for`` carrying only ``lineage``'s operations, with no 

67 loop-carried values. 

68 :raises PassFailedException: If ``op`` is not an ``scf.for``, if the loop has an 

69 unsupported shape, or if the copy would have to carry a loop-carried value. 

70 """ 

71 if not isinstance(op, ForOp): 

72 raise PassFailedException( 

73 f"{op.name} encloses more than one frame lineage. Only scf.for shot loops " 

74 "are split into per-sequencer copies; splitting this operation would need " 

75 "rules for its regions and block arguments that nothing yet exercises." 

76 ) 

77 _reject_unsupported_shape(op) 

78 return _build_loop(op, _retained_ops(op, lineage, analysis), lineage, value_mapper)

81def _reject_unsupported_shape(op: ForOp) -> None: 

82 """Reject shot loops whose shape cannot be split into one copy per frame. 

83 

84 Three shapes are rejected: 

85 

86 * a frame carried between iterations as a loop-carried value; 

87 * a region-bearing operation nested in the loop body; 

88 * an acquisition consumed by anything other than ``pulse.integrate`` or results 

89 bookkeeping. 

90 

91 Each is rejected rather than handled because splitting it needs a rule nothing 

92 currently exercises, and because getting it wrong is silent: the copy would simply not 

93 contain the work, leaving valid IR that describes a different program. 

94 

95 :param op: Shot loop whose shape is checked. 

96 :raises PassFailedException: If ``op`` has any of the shapes listed above. 

97 """ 

98 block = op.body.block 

99 

100 # Frames carried between iterations. The analysis cannot resolve a frame arriving as 

101 # a block argument, so ownership inside the body would be unknown, and a frame-valued 

102 # scf.yield would collide with the terminator the copy builds. 

103 for index, argument in enumerate(block.args[1:], start=1): 

104 if isinstance(argument.type, FrameType): 

105 raise PassFailedException( 

106 f"{op.name} carries a frame as loop-carried value {index}; frames " 

107 "crossing iterations cannot be split per sequencer" 

108 ) 

109 

110 # Nested regions. Retained operations are copied from the body block only, so work 

111 # inside a nested region would be dropped rather than copied. 

112 for inner in block.ops: 

113 if inner.regions: 

114 raise PassFailedException( 

115 f"{inner.name} nested inside a shot loop is region-bearing; fission " 

116 "copies the loop body only and would silently drop its contents" 

117 ) 

118 

119 # Acquisition consumers. pulse.integrate is carried into the copy and results 

120 # bookkeeping is dropped deliberately, but any other consumer would be dropped 

121 # silently and the acquisition would come to mean something different. 

122 for inner in block.ops: 

123 if not isinstance(inner, AcquireOp): 

124 continue 

125 for use in inner.acquisition_result.uses: 

126 consumer = use.operation 

127 if isinstance(consumer, IntegrateOp) or is_results_op(consumer): 

128 continue 

129 raise PassFailedException( 

130 f"{inner.name} result is consumed by {consumer.name}; an outlined " 

131 "sequence carries only pulse.integrate, so this consumer would be lost" 

132 )

135def _retained_ops( 

136 op: Operation, lineage: FrameLineage, analysis: FrameLineageAnalysis 

137) -> set[Operation]: 

138 """Return the operations inside ``op`` that ``lineage``'s copy must contain. 

139 

140 Seeded with the operations the analysis attributes to ``lineage``, then closed over 

141 operands so the copy is self-contained. ``pulse.integrate`` is pulled in explicitly: 

142 it is a downstream marker for its acquisition rather than an operand dependency, and 

143 nothing else would reach it. 

144 """ 

145 retained: set[Operation] = set() 

146 pending = [inner for inner in op.walk() if lineage in analysis.lineages_for_op(inner)] 

147 while pending: 

148 current = pending.pop() 

149 if current in retained: 

150 continue 

151 retained.add(current) 

152 if isinstance(current, AcquireOp): 

153 pending.extend( 

154 use.operation 

155 for use in current.acquisition_result.uses 

156 if isinstance(use.operation, IntegrateOp) 

157 ) 

158 for operand in current.operands: 

159 owner = operand.owner 

160 if isinstance(owner, Operation) and owner is not op and op.is_ancestor(owner): 

161 pending.append(owner) 

162 return retained

165def _build_loop( 

166 op: ForOp, 

167 retained: set[Operation], 

168 lineage: FrameLineage, 

169 value_mapper: dict[SSAValue, SSAValue], 

170) -> ForOp: 

171 """Build a loop over ``retained``, carrying no values between iterations.""" 

172 block = op.body.block 

173 for index, argument in enumerate(block.args[1:], start=1): 

174 if any(use.operation in retained for use in argument.uses): 

175 raise PassFailedException( 

176 f"{op.name} carries value {index} into operations retained for " 

177 f"{lineage.port}, which a hardware sequence cannot express" 

178 ) 

179 

180 body = Block(arg_types=[block.args[0].type]) 

181 mapper = dict(value_mapper) 

182 mapper[block.args[0]] = body.args[0] 

183 body.add_ops([inner.clone(mapper) for inner in block.ops if inner in retained]) 

184 body.add_op(YieldOp()) 

185 

186 bounds = [value_mapper.get(bound, bound) for bound in (op.lb, op.ub, op.step)] 

187 for result in op.results: 

188 value_mapper.pop(result, None) 

189 return ForOp(*bounds, [], body)
