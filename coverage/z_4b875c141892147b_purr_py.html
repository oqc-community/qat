39def _extract_passive_reset_time( 

40 model: CanonicalSystemData | None, 

41 fallback_reset_time: float, 

42) -> float: 

43 """Resolve the passive reset duration for target data. 

44 

45 Canonical reset-method durations are stored in picoseconds. If the canonical model 

46 defines a passive reset duration, convert it back to seconds and return it. Otherwise, 

47 fall back to the builder model's default repetition period. 

48 """ 

49 

50 if model is None: 

51 return fallback_reset_time 

52 

53 for reset_method in model.reset_methods: 

54 if reset_method.type != "passive": 

55 continue 

56 

57 for attribute in reset_method.attributes: 

58 if attribute.key == "duration" and isinstance(attribute.value, int | float): 

59 return float(attribute.value) * _PICOSECONDS_AS_SECONDS 

60 

61 return fallback_reset_time

64def _build_pass_manager( 

65 hardware_model: QuantumHardwareModel, 

66 passive_reset_time: float, 

67) -> PassManager: 

68 """Build the PuRR sanitisation pipeline used before the PuRR import. 

69 

70 :param hardware_model: Hardware model used by analysis and sanitisation passes. 

71 :returns: A pass manager configured to materialise legacy implicit PuRR behaviour before 

72 import. 

73 """ 

74 

75 target_data = TargetData( 

76 default_shots=hardware_model.default_repeat_count, 

77 QUBIT_DATA=QubitDescription(passive_reset_time=passive_reset_time), 

78 ) 

79 return ( 

80 PassManager() 

81 | ActivePulseChannelAnalysis(hardware_model) 

82 | RepeatSanitisation(hardware_model, target_data) 

83 | ReturnSanitisation() 

84 | AcquireSanitisation() 

85 | InitialPhaseResetSanitisation() 

86 | EndOfTaskResetSanitisation() 

87 | ResetsToDelays(target_data) 

88 | SynchronizeTask() 

89 )

92class PurrFrontend(BaseFrontend): 

93 """Import PuRR builders into pulse-level IR after legacy sanitisation. 

94 

95 PuRR IR built using :class:`QuantumInstructionBuilder` can omit structure that legacy 

96 execution paths insert implicitly. This frontend reconstructs that structure before 

97 importing to the new IR stack. The sanitisation pipeline can add: 

98 

99 * the shots loop via a Repeat instruction, if not provided, 

100 * end-of-line resets on the qubits used, 

101 * a return instruction if not provided, 

102 * start-of-line phase accumulation resets. 

103 

104 This keeps the importer focused on translation while preserving legacy program 

105 behaviour expected by existing PuRR workloads. 

106 """ 

107 

108 def __init__( 

109 self, model: CanonicalSystemData | None = None, run_purr_pipeline: bool = True 

110 ): 

111 """ 

112 :param model: The model as a :class:`CanonicalSystemData` representation. Optional, 

113 but required if post-selection is enabled. The frontend derives a 

114 post-selection view from this model and applies it to the mapped results. 

115 :param run_purr_pipeline: Can be used to specify if the pipeline should run before 

116 importing. 

117 """ 

118 self.model = model 

119 self._run_pipeline = run_purr_pipeline 

120 

121 def emit( 

122 self, 

123 src: QuantumInstructionBuilder, 

124 res_mgr: ResultManager | None = None, 

125 met_mgr: MetricsManager | None = None, 

126 compiler_config: CompilerConfig | None = None, 

127 **kwargs, 

128 ) -> ModuleOp: 

129 """Build a :class:`ModuleOp` from a PuRR instruction builder. 

130 

131 :param src: The legacy PuRR representation of quantum programs. 

132 :param res_mgr: Collection of analysis results with caching and aggregation 

133 capabilities, defaults to None. 

134 :param met_mgr: Stores useful intermediary metrics that are generated during 

135 compilation, defaults to None. 

136 :param compiler_config: Compiler settings, defaults to None. 

137 :returns: A compiler :class:`ModuleOp` that holds the program in pulse-level 

138 representation. 

139 :raises TypeError: If *src* is not a :class:`QuantumInstructionBuilder`. 

140 :raises ValueError: If post-selection is enabled but no canonical model was 

141 provided at frontend construction. 

142 """ 

143 

144 if not isinstance(src, QuantumInstructionBuilder): 

145 raise TypeError( 

146 f"PuRR frontend does not support object of type `{type(src).__name__}`." 

147 ) 

148 

149 res_mgr, met_mgr, compiler_config = self._check_metrics_and_config( 

150 res_mgr, met_mgr, compiler_config 

151 ) 

152 

153 # Use the builder's legacy hardware model for sanitisation. The canonical model, 

154 # when provided, supplies passive reset metadata for target data and post-selection. 

155 hardware_model = src.model 

156 passive_reset_time = _extract_passive_reset_time( 

157 self.model, hardware_model.default_repetition_period 

158 ) 

159 

160 if self._run_pipeline: 

161 pipeline = _build_pass_manager(hardware_model, passive_reset_time) 

162 src = pipeline.run( 

163 src, 

164 res_mgr, 

165 met_mgr, 

166 compiler_config=compiler_config, 

167 **kwargs, 

168 ) 

169 

170 if compiler_config.post_selection: 

171 if self.model is None: 

172 raise ValueError( 

173 "Canonical system data description must be provided if post-selection " 

174 "is enabled." 

175 ) 

176 

177 pp_derived_view = PostProcessing.derive(self.model) 

178 post_selection_builder = PostSelectionBuilder(pp_derived_view, enabled=True) 

179 else: 

180 post_selection_builder = None 

181 

182 return PurrImporter(post_selection_builder).build(src)
