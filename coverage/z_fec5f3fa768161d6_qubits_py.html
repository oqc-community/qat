20def _measurement_fidelity_from_readout( 

21 readout: ReadoutProbabilityData | None, 

22) -> float | None: 

23 """Compute scalar measurement fidelity from readout confusion probabilities. 

24 

25 .. math:: 

26 

27 F = \\frac{1}{2}\\bigl[P(0|0) + P(1|1)\\bigr] 

28 

29 where :math:`P(m|p)` is the probability of measuring state :math:`m` given that 

30 state :math:`p` was prepared. Equivalent to 

31 :math:`1 - \\frac{1}{2}[P(1|0) + P(0|1)]` when each row sums to 1. 

32 

33 :param readout: Readout confusion data, or ``None``. 

34 :returns: Fidelity scalar, or ``None`` if data is absent or has no diagonal entries. 

35 """ 

36 if readout is None: 

37 return None 

38 diagonal = tuple( 

39 e for e in readout.probability_entries if e.prepared_state == e.measured_state 

40 ) 

41 return sum(e.probability for e in diagonal) / len(diagonal) if diagonal else None

44def _flatten_operation_ids(operation: OperationData) -> tuple[str, ...]: 

45 """Recursively collect ids of leaf operations within an operation tree, depth-first. 

46 

47 Walks the ``operation_steps`` of each variant. Any step that is itself an 

48 :class:`~qat.experimental.system_data.canonical.schema.OperationData` is recursed 

49 into. A node is considered a leaf when none of its steps are ``OperationData``; 

50 at that point its ``id`` is collected. 

51 

52 :param operation: Root operation to flatten. 

53 :returns: Tuple of leaf operation ids in depth-first traversal order. 

54 """ 

55 all_steps = [step for variant in operation.variants for step in variant.operation_steps] 

56 nested_ops = [s for s in all_steps if isinstance(s, OperationData)] 

57 if not nested_ops: 

58 return (operation.id,) 

59 ids: list[str] = [] 

60 for nested in nested_ops: 

61 ids.extend(_flatten_operation_ids(nested)) 

62 return tuple(ids)

113@dataclass(frozen=True) 

114class QubitView(DerivedViewInterface[CanonicalSystemData]): 

115 """Derived qubit-level view for gate-level compilation. 

116 

117 By default all qubits in the canonical data are included. Pass ``qubit_ids`` 

118 to :meth:`derive` to restrict the view to a specific subset; couplings are 

119 automatically filtered to pairs where both endpoints are within that subset. 

120 

121 :ivar qubits: Per-qubit properties keyed by qubit identifier. 

122 :ivar interactions: Directed interactions between qubits, with positions resolved 

123 against :attr:`qubits` iteration order and gate fidelities pre-extracted. 

124 """ 

125 

126 qubits: Mapping[str, QubitProperties] 

127 interactions: tuple[Interaction, ...] 

128 

129 @classmethod 

130 def derive( 

131 cls, 

132 parent: CanonicalSystemData, 

133 qubit_ids: set[str] | None = None, 

134 **kwargs, 

135 ) -> QubitView: 

136 """Constructs a qubit view over a subset of qubits in the canonical data. 

137 

138 When ``qubit_ids`` is ``None`` all qubits are included. Interactions are filtered 

139 to pairs where both endpoints are within the included set. 

140 

141 :param parent: Canonical system data. 

142 :param qubit_ids: Optional set of qubit ids to include. Defaults to all 

143 qubits when ``None``. 

144 :returns: :class:`QubitView` restricted to the given qubit ids. 

145 """ 

146 selected = cls._select_qubits(parent, qubit_ids) 

147 qubits = cls._unpack_qubit_properties(selected) 

148 position_by_id = {qid: i for i, qid in enumerate(qubits)} 

149 interactions = cls._unpack_interactions(parent, position_by_id) 

150 return cls( 

151 qubits=MappingProxyType(qubits), 

152 interactions=interactions, 

153 )

155 @staticmethod 

156 def _select_qubits( 

157 parent: CanonicalSystemData, 

158 qubit_ids: set[str] | None, 

159 ) -> dict: 

160 """Returns the selected canonical qubit records in canonical order. 

161 

162 Warns and drops any requested ids not present in the canonical data. 

163 

164 :param parent: Canonical system data. 

165 :param qubit_ids: Optional set of qubit ids to include. ``None`` means all. 

166 :returns: Dict of canonical qubit records keyed by id, in canonical order. 

167 """ 

168 qubit_by_id = {q.id: q for q in parent.qubits} 

169 if qubit_ids is None: 

170 return qubit_by_id 

171 unknown = qubit_ids - qubit_by_id.keys() 

172 if unknown: 

173 warnings.warn( 

174 f"Requested qubit ids not found in canonical data and will be " 

175 f"ignored: {sorted(unknown)}", 

176 stacklevel=3, 

177 ) 

178 # Preserve canonical ordering for deterministic positions. 

179 return {q.id: q for q in parent.qubits if q.id in qubit_ids} 

180 

181 @staticmethod 

182 def _unpack_qubit_properties(selected: dict) -> dict[str, QubitProperties]: 

183 """Unpacks per-qubit properties from the selected canonical qubit records. 

184 

185 :param selected: Canonical qubit records keyed by id. 

186 :returns: Dict of :class:`QubitProperties` keyed by qubit id. 

187 """ 

188 return { 

189 qid: QubitProperties( 

190 index=qubit.index, 

191 supported_operations=OperationSet( 

192 operation_type=tuple( 

193 op_id 

194 for operation in qubit.operations 

195 for op_id in _flatten_operation_ids(operation) 

196 ) 

197 ), 

198 measurement_fidelity=_measurement_fidelity_from_readout( 

199 qubit.readout_probability 

200 ), 

201 ) 

202 for qid, qubit in selected.items() 

203 } 

204 

205 @staticmethod 

206 def _unpack_interactions( 

207 parent: CanonicalSystemData, 

208 position_by_id: dict[str, int], 

209 ) -> tuple[Interaction, ...]: 

210 """Unpacks sorted interactions from canonical couplings filtered to included qubits. 

211 

212 :param parent: Canonical system data. 

213 :param position_by_id: Map from qubit id to its position in the view. 

214 :returns: Tuple of :class:`Interaction` sorted by ``(source_position, target_position)``. 

215 """ 

216 return tuple( 

217 sorted( 

218 ( 

219 Interaction( 

220 source_position=position_by_id[c.source_qubit_id], 

221 target_position=position_by_id[c.target_qubit_id], 

222 gate_fidelities=MappingProxyType( 

223 {gf.gate: float(gf.fidelity) for gf in c.gate_fidelities} 

224 ), 

225 ) 

226 for c in parent.couplings 

227 if c.source_qubit_id in position_by_id 

228 and c.target_qubit_id in position_by_id 

229 ), 

230 key=lambda i: (i.source_position, i.target_position), 

231 ) 

232 )
